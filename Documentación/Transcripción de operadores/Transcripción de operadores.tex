	\section{Operadores de Comparación}
	
		\begin{tabular}{|c|>{\centering\arraybackslash}p{5cm}|p{8cm}|}
			\hline
				\textbf{C} & \textbf{Circom} & \textbf{Comentario} \\
			\hline
				\texttt{a == b} & \texttt{IsZero()(a - b)} & Si \(a = b\), entonces se cumple (==1) \\
			\hline
				\texttt{a != b} & \texttt{1- IsZero()(a - b)} & Si \(a \neq b\), entonces es falso (==0), por tanto se invierte con la resta \\
			\hline
				\texttt{\(a < b\)} & LessThan(n)(a,b) & Si \(a < b\) es verdadero, entonces se cumple (==1) \\
			\hline
				\texttt{\(a \leq b\)} & LessThanEq(n)(a,b) & Si \(a \leq b\) es verdadero, entonces se cumple (==1) \\
			\hline
				\texttt{\(a > b\)} & 1- LessThanEq(n)(a,b) & Si \(a \leq b\) es falso (== 0), entonces \(a > b\) \\
			\hline
				\texttt{\(a \geq b \)} & 1- LessThan(n)(a,b) & Si \(a < b\) es falso (== 0), entonces \(a \geq b \) \\
			\hline
		\end{tabular}
		
		\smallskip
		\noindent \textit{Nota:} Las funciones \texttt{IsZero()} son circuitos que comparan si un valor es igual a 0.
		
		\noindent \textit{Nota:} Las funciones \texttt{LessThan(n)} y \texttt{LessThanEq(n)} son circuitos que comparan dos valores (a < b) de hasta \( n \) bits.
		
		\section{Operadores Aritméticos}
		
		\noindent Tanto en C como en Circom, los operadores suma (+), resta (-) y multiplicación (*) no difieren. En cambio, en operaciones como la división entera (/) y el módulo (\%) si que encontramos diferencias y por ende es necesaria su transcripción.
		
		\vspace{0.3cm}
		
		\noindent La idea desde la que se parte para encontrar esta equivalencia es de la propiedad fundamental de la división entera, la cual establece:
		
		\[
		\scalebox{1.2}{$a = b \cdot q + r$}
		\]
		
		\vspace{0.3cm}
		
		\subsection*{Operación división}
		
		\noindent Partiendo de la fórmula \(a = b * q + r\), podemos llegar al resultado de la operación \(a / b\) forzando a que se cumpla la fórmula pero despreciando el resto (ya que se trata de división entera).
		
		\vspace{0.3cm}
		
		\noindent El término "forzar" hace referencia a usar una variable en Circom, de forma que partiendo del dividendo y del divisor, el cociente solo pueda tener ese valor.
		
		
		Ejemplo de código en Circom:
		
		\begin{lstlisting}[language=Circom]
			template IntegerDivision() {
				signal input a; // Dividendo
				signal input b; // Divisor
				signal output c; // Cociente q
				
				var q;
				a === b * q;
				c <== q;
			}
			
			component main = IntegerDivision();
		\end{lstlisting}
		
		\subsection*{Operación módulo}
		
		\noindent Partiendo de la fórmula \(a = b * q + r\), podemos llegar al resultado de la operación \(a \% b\) con la variable r, la que representa el resto de una división.
		
		\begin{align}
			a &= b \cdot q + r \\
			r &= a - (b \cdot q) \\
			r &= a - (b \cdot ( a / b) )
		\end{align}
		
		\noindent \textit{Nota: } q representa el resultado de una división, por lo que podemos decir que \( q = a/b \).
		
		\vspace{0.4cm}
		
		\begin{tabular}{|c|>{\centering\arraybackslash}p{5cm}|p{8cm}|}
			\hline
			\textbf{C} & \textbf{Circom} & \textbf{Comentario} \\
			\hline
			\texttt{\(a / b\)} & \texttt{ a === b * q} & Forzando q con una variable a que esa ecuación se cumpla \\
			\hline
			\texttt{\(a \% b\)} & \texttt{a - (b * (a/b))} & \(a / b\) hace referencia a la equivalencia que se obtiene en circom para poder realizar la operación \% \\
			\hline
		\end{tabular}
		
		\section{Acceso a Arrays}
		
		\noindent La diferencia principal entre C y Circom en el acceso a arrays radica en que el tamaño de los arrays en Circom ha de ser fijo a la hora de compilar y ser siempre constante, mientras que en C puede ser estático y dinámico. Además, en Circom el valor de los arrays no puede ser reasignado dinámicamente.
		
		\vspace{0.3cm}
		
		\noindent \textit{Nota: En C, el tamaño de un array puede ser variable, ya sea por arrays de longitud variable o por el uso de memoria dinámica}
		
		\vspace{0.3cm}
		% Tabla con codigo C y Circom
		
		\begin{tabular}{p{0.45\textwidth} | p{0.48\textwidth}}
			\hline
			% COdigo en C
			\lstset{language=C}
			\begin{lstlisting}
				// Codigo en C
				#include <stdio.h>
				
				int main() {
					
					// Ejemplo de array con tamanyo variable con funcion f() 
					int n = f();
					int lista[n]; // Se decide en tiempo de ejecucion
					
					
					// EJemplo de array de tamanyo variable con asignacion de memoria dinamica
					int *lista = malloc(x);
					
					//x tamanyo que se quiera reservar en memoria dinamica
				}
			\end{lstlisting} 
			
			&
			% Código en CIrcom
			
			\begin{lstlisting}[language=Circom]
				
				//Codigo circom
				template accesoArray(N) {
					signal input lista[N]; // Tamanyo fijo, suponemos iniciada
					signal output sumaAcumulada;
					
					var suma = 0;
					for(int i = 0; i < N; i++){
						suma = suma + lista[i];
						
						// NO se puede hacer lo siguiente
						// lista[i] <== 3;
					}
					
					sumaAcumulada <== suma;
				}
				
				component main = accesoArray(10);
			\end{lstlisting}
			\\
			\hline
		\end{tabular}
	
		\vspace{0.4cm}
		
		\noindent \textit{Nota: En Circom no se puede fijar el tamaño de un array con una señal.}
		
		\section{Expresiones condicionales}
		
		\noindent En C, al ser un lenguaje de ejecución dinámica, los condicionales pueden decidir el valor final de una variable en tiempo de ejecución.
		
		\vspace{0.3cm}
		
		\noindent En Circom, cada signal representa un “cable” del circuito, que solo puede recibir un valor. Por ello, no es posible reasignar un signal dentro de un condicional dinámico. 
		
		\vspace{0.3cm}
		
		\noindent Para simular decisiones condicionales basadas en entradas (inputs), se deben usar expresiones aritméticas que implementen la lógica de selección, de manera que el valor de salida quede correctamente definido sin reasignar el signal.
		
		\vspace{0.3cm}
		% Tabla con codigo C y Circom
		
		\begin{tabular}{p{0.45\textwidth} | p{0.45\textwidth}}
			\hline
			% COdigo en C
			\lstset{language=C}
			\begin{lstlisting}
				// Codigo en C
				#include <stdio.h>
				
				int main(int argc, char *argv[]) {
					int entrada = argv[1];
					char mensaje[50];
					
					if(entrada % 2 == 0){ 
						sprintf(mensaje, "%d es par\n", entrada);
					}
					else {
						sprintf(mensaje, "%d es impar\n", entrada);
					}
					
					printf("s", mensaje);
					free(mensaje);
					
					return 0;
				}
			\end{lstlisting} 
			
			&
			% Código en CIrcom
			
			\begin{lstlisting}[language=Circom]
				
				//Codigo circom
				template condicionales() {
					signal input entrada;
					signal output salida; //0 par, 1 impar
					
					var var_division;
					entrada == 2 * var_division;
					
					salida <== (entrada - (2*var_division)) 
					
				}
				
				component main = condicionales();
			\end{lstlisting}
			 \\
			 \hline
		\end{tabular}
		
		\vspace{0.3cm}
		
		\noindent Con este ejemplo se ilustra como puede transformarse una expresión condicional que comprueba si un número es par o impar en una expresión aritmética que representa lo mismo y evita el reasignar la señal de salida más de una vez (algo no permitido).
