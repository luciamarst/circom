	\section{Operadores de Comparación}
		\noindent Un detalle a observar en las funciones \textit{LessThan()} y \textit{LessThanEq()} es que para números negativos podría presentar problemas. 
		
		\vspace{0.2cm}
		
		\noindent El funcionamiento de \textit{Num2Bits(n)} cuando recibe un número negativo es el siguiente:
		
		\begin{itemize}
			\item El número negativo se interpreta como un cuerpo finito.
			
			\item Éste se interpretará como un número superior a \(2^n - 1\), saliéndose fuera del rango [0,\(2^n-1\)], por tanto la constraints no se pueden cumplir y la prueba no se generará.
			
			\item En conclusión, con \textit{Num2Bits()} garantizamos el uso de número positivos dentro del rango mencionado de manera implícita.
		\end{itemize}
		
		\noindent \textit{Observación: Gracias a la constraint lc1 === in (fuerza a que la entrada cumpla el rango \([0,2^n -1]\)), podemos evitar el uso de número negativos, ya que estos no cumplirán tal restricción.}
		
		\vspace{5cm}
		
		\textbf{\textit{Num2Bits(n):}}
		
		\begin{lstlisting}[language=Circom]
			template Num2Bits(n){
				signal input in;
				signal output out[n];
				var lc1=0; // Reconstruye el numero decimal a convertir para la constraint
				
				var e2=1; // Evitar la potencia
				for (var i = 0; i<n; i++) {
					out[i] <-- (in >> i) & 1; // Contruimos el numero en binario
					out[i] * (out[i] -1 ) === 0;
					lc1 += out[i] * e2;
					e2 = e2+e2;
				}
				
				lc1 === in; // Constraint mencionada anteriormente
			}
		\end{lstlisting}
		
		\vspace{0.5cm}
		
		\noindent Partiendo de esta información, podemos realizar pequeñas modificaciones en las funciones \textit{LessThan()} y \textit{LessThanEq()} para asegurar el uso de enteros positivos.
		
		\vspace{0.2cm}
		
		\begin{tabular}{|p{0.45\textwidth} | p{0.45\textwidth}|}
			\hline
			\lstset{language=Circom}
			\begin{lstlisting}[language=Circom]
				template SecureLessThan(n){
					signal input in[2];
					signal output out;
					
					//------------------
				
					component checkA = Num2Bits(n);
					checkA.in <== in[0];
					
					component checkB = Num2Bits(n);
					checkB.in <== in[1];
					
					 adder = BinSum(n, 2);
					
					var i;
					for (i=0;i<n;i++) {
						checkA.out[i] ==> adder.in[0][i];
						checkB.out[i] ==> adder.in[1][i];
					}
					
					// Miramos el bit + significativo (MSB), 1 si A < B, 0 de lo contrario
					adder.out[n-1] ==> out;
				}
			\end{lstlisting} 
			
			&
			% Código en CIrcom
			
			\begin{lstlisting}[language=Circom]
				template SecureLessThanEq(n){
					signal input in[2];
					signal output out;
					
					//----------------------
					component checkA = Num2Bits(n);
					checkA.in <== in[0];
					
					component checkB = Num2Bits(n);
					checkB.in <== in[1];
					
					adder = BinSum(n, 2);
					
					var i;
					for (i=0;i<n;i++) {
						checkA.out[i] ==> adder.in[0][i];
						checkB.out[i] ==> adder.in[1][i];
					}
					
					// Miramos el bit + significativo (MSB), 1 si A < B, 0 de lo contrario
					component equal = IsEqual();
					equal.in[0] <== in[0];
					equal.in[1] <== in[1]; 
					
					out <==	adder.out[n-1] + equal.out - adder.out[n-1] * equal.out;
				}
				
			\end{lstlisting}
			\\
			\hline
		\end{tabular}
			
		\vspace{0.3cm}	
		
		\noindent Con está pequeña modificación, al usar para ambos número a comparar la función \textit{Num2Bits()} nos aseguramos de que se cumpla la restricción de ser número enteros positivos dentro del rango.
		
		\vspace{0.3cm}	
		
		\begin{tabular}{|c|>{\centering\arraybackslash}p{5cm}|p{8cm}|}
			\hline
				\textbf{C} & \textbf{Circom} & \textbf{Comentario} \\
			\hline
				\texttt{a == b} & \texttt{IsZero()(a - b)} & Si \(a = b\), entonces se cumple (==1) \\
			\hline
				\texttt{a != b} & \texttt{1- IsZero()(a - b)} & Si \(a \neq b\), entonces es falso (==0), por tanto se invierte con la resta \\
			\hline

			\hline

		\section{Operadores Aritméticos}
		
		\noindent Tanto en C como en Circom, los operadores suma (+), resta (-) y multiplicación (*) no difieren. En cambio, en operaciones como la división entera (/) y el módulo (\%) si que encontramos diferencias y por ende es necesaria su transcripción.
		
		\vspace{0.3cm}
		
		\noindent La idea desde la que se parte para encontrar esta equivalencia es de la propiedad fundamental de la división entera, la cual establece:
		
		\[
		\scalebox{1.2}{$a = b \cdot q + r$}
		\]
		
		\vspace{0.3cm}
		
		\subsection*{Operación división}
		
		\noindent Partiendo de la fórmula \(a = b * q + r\), podemos llegar al resultado de la operación \(a / b\) forzando a que se cumpla la fórmula pero despreciando el resto (ya que se trata de división entera).
		
		\vspace{0.3cm}
		
		\noindent El término "forzar" hace referencia a usar una variable en Circom, de forma que partiendo del dividendo y del divisor, el cociente solo pueda tener ese valor.
		
		
		Ejemplo de código en Circom:
		
		\begin{lstlisting}[language=Circom]
			template IntegerDivision() {
				signal input a; // Dividendo
				signal input b; // Divisor
				signal output c; // Cociente q
				
				var q;
				a === b * q;
				c <== q;
			}
			
			component main = IntegerDivision();
		\end{lstlisting}
		
		\subsection*{Operación módulo}
		
		\noindent Partiendo de la fórmula \(a = b * q + r\), podemos llegar al resultado de la operación \(a \% b\) con la variable r, la que representa el resto de una división.
		
		\vspace{0.2cm}
		
		\noindent Por otro lado, ha de cumplirse la expresión \(0 \leq r < b\). De lo contrario, podría producirse problemas en cuanto a la seguridad del circuito. 
		
		\begin{align}
			a &= b \cdot q + r \\
			r &= a - (b \cdot q) \\
			r &= a - (b \cdot ( a / b) )
		\end{align}
		
		\noindent \textit{Nota: } q representa el resultado de una división, por lo que podemos decir que \( q = a/b \).
		
		\vspace{0.4cm}
		
		\noindent Para que se cumpla la restricción \(0 \leq r < b\) :
		
		\begin{lstlisting}[language=Circom]
			component lessThan = SecureLessThan(n);
			lessThan.in[0] <== r;
			lessThan.in[1] <== b;
			
			// 1 si r < b, como ya se checkea en SecureLessThan que sea positivo o 0 es suficiente
			
			lessThan.out === 1 // Hacemos que se cumpla la restriccion,
			// Si no se cumple, no se generara la prueba
		\end{lstlisting}
		
		\vspace{0.2cm}
		
		\begin{tabular}{|c|>{\centering\arraybackslash}p{5cm}|p{8cm}|}
			\hline
			\textbf{C} & \textbf{Circom} & \textbf{Comentario} \\
			\hline
			\texttt{\(a / b\)} & \texttt{a === b * q + r} & Forzando q con una variable a que esa ecuación se cumpla \\
			\hline
			\texttt{\(a \% b\)} & \texttt{a - (b * (a/b))} & \(a / b\) hace referencia a la equivalencia que se obtiene en circom para poder realizar la operación \% \\
			\hline
		\end{tabular}
		
		\section{Acceso a Arrays}
		
		\noindent La diferencia principal entre C y Circom en el acceso a arrays radica en que el tamaño de los arrays en Circom ha de ser fijo a la hora de compilar y ser siempre constante, mientras que en C puede ser estático y dinámico. Además, en Circom el valor de los arrays no puede ser reasignado dinámicamente.
		
		\vspace{0.3cm}
		
		\noindent \textit{Nota: En C, el tamaño de un array puede ser variable, ya sea por arrays de longitud variable o por el uso de memoria dinámica}
		
		\vspace{0.3cm}
		% Tabla con codigo C y Circom
		
		\begin{tabular}{p{0.48\textwidth} | p{0.48\textwidth}}
			\hline
			% COdigo en C
			\lstset{language=C}
			\begin{lstlisting}
				// Codigo en C
				#include <stdio.h>
				
				int main() {
					
					// Ejemplo de array con tamanyo variable con funcion f() 
					int n = f();
					int lista[n]; // Se decide en tiempo de ejecucion
					
					
					// EJemplo de array de tamanyo variable con asignacion de memoria dinamica
					int *lista = malloc(x);
					
					//x tamanyo que se quiera reservar en memoria dinamica
				}
			\end{lstlisting} 
			
			&
			% Código en CIrcom
			
			\begin{lstlisting}[language=Circom]
				
				//Codigo circom
				template accesoArray(N) {
					signal input lista[N]; // Tamanyo fijo, suponemos iniciada
					signal output sumaAcumulada;
					
					var suma = 0;
					for(int i = 0; i < N; i++){
						suma = suma + lista[i];
						
						// NO se puede hacer lo siguiente
						// lista[i] <== 3;
					}
					
					sumaAcumulada <== suma;
				}
				
				component main = accesoArray(10);
			\end{lstlisting}
			\\
			\hline
		\end{tabular}
	
		\vspace{0.4cm}
		
		\noindent \textit{Nota: En Circom no se puede fijar el tamaño de un array con una señal.}
		
		\vspace{2.6cm}
		
		\noindent En Circom, no se puede acceder directamente a posiciones de arrays con señales inputs. Sin embargo, se puede acceder indirectamente a esta posición sin usar esta señal input.
		
		\vspace{0.4cm}
			
		\begin{lstlisting}[language=Circom]
			signal input pos;
			signal input lista[N]; 
			signal output acceso;
			
			// NO SE PUEDE HACER
			//lista[pos];
			
			// SI SE PUEDE HACER, coste O(N)
			// Hay otras formas como arboles que reducen el coste a O(log (N))
			for(int i = 0; i < N; i++){
				signal seleccionado;
				seleccionado <== (i === pos ? 1 : 0);
				acceso <== acceso + lista[i] * seleccionado;
			}
			
		\end{lstlisting}
		
		\section{Expresiones condicionales}
		
		\noindent En C, al ser un lenguaje de ejecución dinámica, los condicionales pueden decidir el valor final de una variable en tiempo de ejecución.
		
		\vspace{0.3cm}
		
		\noindent En Circom, cada signal representa un “cable” del circuito, que solo puede recibir un valor. Por ello, no es posible reasignar un signal dentro de un condicional dinámico. 
		
		\vspace{0.3cm}
		
		\noindent Para simular decisiones condicionales basadas en entradas (inputs), se deben usar expresiones aritméticas que implementen la lógica de selección, de manera que el valor de salida quede correctamente definido sin reasignar el signal.
		
		\vspace{0.3cm}
		% Tabla con codigo C y Circom
		
		\noindent Un ejemplo específico de conversión de un condicional C a uno en Circom puede ser el siguiente: 
		
		\vspace{0.3cm}
		
		\begin{tabular}{p{0.45\textwidth} | p{0.45\textwidth}}
			\hline
			% COdigo en C
			\lstset{language=C}
			\begin{lstlisting}
				// Codigo en C
				#include <stdio.h>
				
				int main(int argc, char *argv[]) {
					int entrada = argv[1];
					char mensaje[50];
					
					if(entrada % 2 == 0){ 
						sprintf(mensaje, "%d es par\n", entrada);
					}
					else {
						sprintf(mensaje, "%d es impar\n", entrada);
					}
					
					printf("s", mensaje);
					free(mensaje);
					
					return 0;
				}
			\end{lstlisting} 
			
			&
			% Código en CIrcom
			
			\begin{lstlisting}[language=Circom]
				
				//Codigo circom
				template condicionales() {
					signal input entrada;
					signal output salida; //0 par, 1 impar
					
					var var_division;
					entrada == 2 * var_division;
					
					salida <== (entrada - (2*var_division)) 
					
				}
				
				component main = condicionales();
			\end{lstlisting}
			 \\
			 \hline
		\end{tabular}
		
		\vspace{0.3cm}
		
		\noindent Con este ejemplo se ilustra como puede transformarse una expresión condicional que comprueba si un número es par o impar en una expresión aritmética que representa lo mismo y evita el reasignar la señal de salida más de una vez (algo no permitido).
		
		\noindent A la hora de generalizar esta conversión condicional de C a Circom, primero tenemos que darnos cuenta del patrón que se sigue partiendo desde el caso más simple.
		
		\subsection*{Caso más simple: if/else}
		
		\begin{lstlisting}[language=C]
			
			// Codigo en C
			if (age < 0) {
				printf("Edad invalida.\n");
			} else {
				printf("Perfil: adulto mayor.\n");
			}
		\end{lstlisting} 
		
		\begin{lstlisting}[language=Circom]
			
			// Codigo en Circom
			component less_0 = LessThan()(age,0);
			
			signal less_0 = less_0.out;
			signal not_less_0 = 1 - less_0.out;
			
			mensaje <== less_0 * m1 + not_less_0 * m2;
		\end{lstlisting}
		
		\vspace{0.2cm}
		
		\noindent Como podemos ver, el patrón que sigue es crear una componente equivalente a la expresión lógica utilizada en el if. Con esta componente, al tener dos opciones, el if o el else, observamos que el else siempre será la suma de la expresiones anteriores menos 1, ya que al no cumplir las anteriores tiene que ser lo contrario.
		
		\vspace{0.2cm}
		
		\noindent Este caso es demasiado simple, por lo que un ejemplo con más condicionales detallará mejor lo explicado anteriormente.
		
		\subsection*{Caso complejo: if/else if/else}
			\begin{lstlisting}[language=C]
				
				// Codigo en C
				
				if (age < 0) {
					printf("Edad invalida.\n");
				} else if (age < 13) {
					printf("Perfil: menor.\n");
				} else if (age < 18) {
					printf("Perfil: adolescente.\n");
				} else if (age < 30) {
					printf("Perfil: joven adulto.\n");
				} else if (age < 60) {
					printf("Perfil: adulto.\n");
				} else {
					printf("Perfil: adulto mayor.\n");
				}
			\end{lstlisting}
			
			
			\begin{itemize}
				\item Por cada condición de igualdad/desigualdad se crea un componente correspondiente:
				\begin{lstlisting}[language=Circom]
					component less_0 = LessThan()(age,0);
					component less_13 = LessThan()(age,13);
					component less_18 = LessThan()(age,18);
					component less_30 = LessThan()(age,30);
					component less_60 = LessThan()(age,60);
				\end{lstlisting}
				
				\item Se generan señales intermedias para que las condiciones sean mutuamente excluyentes:
				\begin{lstlisting}[language=Circom]
					signal is_less_0 = less_0.out;
					signal is_less_13 = (1 - less_0.out) * less_13.out;
					signal is_less_18 = (1 - less_0.out) * (1 - less_13.out) * less_18.out;
					signal is_less_30 = (1 - less_0.out) * (1 - less_13.out) * (1 - less_18.out) * less_30.out;
					signal is_less_60 = (1 - less_0.out) * (1 - less_13.out) * (1 - less_18.out) * (1 - less_30.out) * less_60.out;
					signal not_less_60 = 1 - less_60.out;
				\end{lstlisting}
				
				\item Combinación ponderada de los mensajes:
				\begin{lstlisting}[language=Circom]
					mensaje <== is_less_0 * m1 +
					is_less_13 * m2 +
					is_less_18 * m3 +
					is_less_30 * m4 +
					is_less_60 * m5 +
					not_less_60 * m6;
				\end{lstlisting}
			\end{itemize}
			
			\noindent Como podemos observar, siempre se repite el mismo patrón en el que cada condición utiliza la negación de las condiciones anteriores y el resultado de la comparación actual para el caso de los \textit{else/if}
			
			\vspace{0.2cm}
			
			\noindent En cuanto al \textit{else}, como se explico en el caso simple, se observa que siempre es la negación de la suma de todas las condiciones anteriores, quedando reflejado con más claridad en este ejemplo 
			
			\vspace{0.2cm}
			
			\noindent Para el caso del \textit{switch} sería similar, ya que es una estructura semejante a la de  \textit{if/else if/else}.
			
			\vspace{0.3cm}
				
			\noindent \textit{Observaciones: En este ejemplo concreto se uso el operador lógico <, pero para cualquier otra operación lógica bastaría con aplicar la transformación especificada en el apartado de operadores lógicos y de comparación.} 
		
		
		\subsection*{Condicionales anidados}
				\begin{lstlisting}[language=C]
				
				// Codigo en C
				
				if (age < 20) {
					if(age < 10){
						printf("Perfil: ninyo.\n");
					}
					else{
						printf("Perfil: adolescente.\n");
					}
				} else if(age < 40){
					if(age < 30){
						printf("Perfil: adulto joven.\n");
					}
					else {
						printf("Perfil: adulto mayor.\n");
					}
				}
				else{
					printf("Perfil: adulto mayor.\n");
				}
			\end{lstlisting}
			
			
			\begin{itemize}
				\item Por cada condición de igualdad/desigualdad se crea un componente correspondiente, como en los casos anteriores.
				
				\begin{lstlisting}[language=Circom]
					component less_10 = LessThan()(age,10);
					component less_20 = LessThan()(age,20);
					component less_30 = LessThan()(age,30);
					component less_40 = LessThan()(age,40);
				\end{lstlisting}
				
				\item Partiendo de lo citado an los apartados anteriores, tenemos que generar las condiciones excluyentes.
				
				\begin{lstlisting}[language=Circom]
					signal is_less_20 = less_20.out;
					signal is_less_10 = is_less_20 * less_10.out;
					signal isnt_less_10 = is_less_20 * (1 - less_10.out);
					
					signal is_less_40 = less_40.out;
					signal is_less_30 = is_less_40 * less_30.out;
					signal isnt_less_30 = is_less_40 * (1 - less_30.out);
					
					
					signal is_greater_40 = (1-is_less_40);
				\end{lstlisting}
				
				\item Combinación ponderada de los mensajes:
				\begin{lstlisting}[language=Circom]
					mensaje <== is_less_10 * m1 + isnt_less_10 * m2 +
					is_less_30 * m3 + isnt_less_30 * m4 +
					is_greater_40 * m5;
				\end{lstlisting}
			\end{itemize}
		
		\section{Operadores de desplazamiento}
		
		
		\noindent Los operadores de desplazamiento en C funcionan de la siguiente manera:
		
		\vspace{0.3cm}
		
		\noindent Imaginemos que tenemos el número 2 que queremos desplazar 2 bits.
		
		\begin{itemize}
			\item 1. Dado un número decimal 2, transformarlo a binario, que es \texttt{0010}.
			\item 2. Con el n (número de bits a desplazar), desplazamos los dos últimos bits.
			\item \texttt{1000} (8 en decimal) es el número que se obtiene desplazando los bits.
		\end{itemize}
		
		\vspace{0.1cm}
		
		\noindent \textit{Observación: Podemos observar que para obtener el número equivalente del desplazamiento de bits en decimal se consigue con $x * 2^n$ en el caso de desplazamiento a la izquierda y con $x / 2^n$ en el caso de desplazamiento a la derecha. Sin embargo, estas operaciones son muy costosas, por lo que optaremos por otra opción.}
		
		\vspace{0.3cm}
		
		\noindent Para conseguir una equivalencia entre estos operadores, directamente transformamos el número decimal en binario y, en función de si se trata de un desplazamiento a la izquierda o a la derecha, despreciar y añadir por un lado o por otro.
		
		\vspace{0.4cm}
		
		\subsection*{Función Bits2Num()}
		\begin{lstlisting}[language=Circom]
			template Bits2Num(N){
				signal input binario[N];
				signal output decimal;
				
				signal potencia[N];
				signal sum[N];
				
				// Primera potencia = 1
				potencia[0] <== 1;
				sum[0] <== binario[0] * potencia[0];
				
				for (var i = 1; i < N; i++) {
					potencia[i] <== potencia[i-1] * 2;                 // potencia iterativa
					sum[i] <== sum[i-1] + binario[i] * potencia[i];    // acumulacion
				}
				
				decimal <== sum[N-1];
			}
		\end{lstlisting}
		
		
		\subsection*{Desplazamiento a la izquierda}
		\noindent Explicaremos un ejemplo de la aplicación que haremos para \verb|11 << 2|.
		
		\begin{itemize}
			\item Dado un número decimal, lo convertimos a binario, en este caso 11:
			\begin{equation*}
				\texttt{00001011}
			\end{equation*}
			
			\item Con el número en binario, desplazamos 2 bits a la izquierda:
			\begin{equation*}
				\texttt{00001011 → 00101100}
			\end{equation*}
			
			\item Como podemos ver, los últimos dos bits se han transformado en 0, por lo que la idea es añadir un número $n$ de ceros al final del array que representa el número binario.
		\end{itemize}
		
		\begin{lstlisting}[language=Circom]
			// Codigo Circom
			template desplazamiento_izq(N) {
				signal input numero_decimal;
				signal input desplazamiento;
				signal output salida;
				
				
				assert(N >= desplazamiento);
				
				component n2b = Num2Bits(N);
				n2b.in <== numero_decimal;
				
				signal binario[N];
				
				// Desplazar bits a la izquierda
				for (var i = 0; i < N - desplazamiento; i++) {
					binario[i] <== n2b.out[i+desplazamiento];
				}
				
				// Rellenar con ceros al inicio
				for (var i = N - desplazamiento; i < N; i++) {
					binario[i] <== 0;
				}
				
				component b2n = Bits2Num(N);
				b2n.binario <== binario;
				
				salida <== b2n.decimal;
			}
			
			component main = desplazamiento_izq(N);
		\end{lstlisting}
		
		
		\subsection*{Desplazamiento a la derecha}
		\noindent Explicaremos un ejemplo de la aplicación que haremos para \verb|11 >> 2|
		
		\begin{itemize}
			\item Dado un número decimal, lo convertimos a binario, en este caso 11:
			\begin{equation*}
				\texttt{00001011}
			\end{equation*}
			
			\item Con el número en binario, desplazamos 2 bits a la derecha:
			\begin{equation*}
				\texttt{00001011 → 00000010}
			\end{equation*}
			
			\item Como podemos ver, los últimos dos bits se han despreciado, añadiendo dos ceros al principio del número binario.
		\end{itemize}
		
		\begin{lstlisting}[language=Circom]
			// Codigo Circom
			template desplazamiento_dcha(N) {
				signal input numero_decimal;
				signal input desplazamiento;
				signal output salida;
				
				assert(N >= desplazamiento);
				
				component n2b = Num2Bits(N);
				n2b.in <== numero_decimal;
				
				signal binario[N];
				
				// Rellenar con ceros al final
				for (var i = 0; i < desplazamiento; i++) {
					binario[i] <== 0;
				}
				
				// Desplazar bits a la derecha
				for (var i = 0; i < N-desplazamiento; i++) {
					binario[i+desplazamiento] <== n2b.out[i];
				}
				
				component b2n = Bits2Num(N);
				b2n.binario <== binario;
				
				salida <== b2n.decimal;
			}
			
			component main = desplazamiento_dcha(N);
		\end{lstlisting}
		
			\noindent \textit{Observación: Hay que asegurar de que el tamaño de N sea mayor o igual que el desplazamiento que se quiere realizar.}
